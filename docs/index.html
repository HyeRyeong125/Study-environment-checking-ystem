SETUP

설치 및 실행

# 1. 클론
git clone .../Study-environment-checking-system.git

# 2. 의존성
cd backend && pip install -r requirements.txt

# 3. 환경 설정
cp .env.example .env

# 4. 실행
mosquitto & python arduino_gateway.py & python main.py
# → http://localhost:3000
